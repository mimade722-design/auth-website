    <script>
        // توليد النجوم
        function createStars() {
            const stars = document.getElementById('stars');
            for (let i = 0; i < 100; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                star.style.width = Math.random() * 3 + 'px';
                star.style.height = star.style.width;
                star.style.left = Math.random() * 100 + '%';
                star.style.top = Math.random() * 100 + '%';
                star.style.animationDelay = Math.random() * 3 + 's';
                star.style.animationDuration = Math.random() * 2 + 2 + 's';
                stars.appendChild(star);
            }
        }

        // عرض/إخفاء النماذج
        function showLogin() {
            document.getElementById('loginModal').style.display = 'flex';
        }

        function hideLogin() {
            document.getElementById('loginModal').style.display = 'none';
        }

        function showRegister() {
            document.getElementById('registerModal').style.display = 'flex';
        }

        function hideRegister() {
            document.getElementById('registerModal').style.display = 'none';
        }

        function showDashboard(user) {
            document.querySelector('.hero').style.display = 'none';
            document.querySelector('.features').style.display = 'none';
            document.getElementById('dashboard').style.display = 'flex';
            
            if (user) {
                document.getElementById('userName').textContent = user.name;
                document.getElementById('userRole').textContent = user.role;
            }
        }

        // محاكاة تسجيل الدخول
        function simulateLogin() {
            const email = document.getElementById('loginEmail').value;
            const password = document.getElementById('loginPassword').value;
            
            if (!email || !password) {
                showNotification('يرجى ملء جميع الحقول', 'error');
                return;
            }
            
            const user = {
                name: email.split('@')[0],
                email: email,
                role: Math.random() > 0.7 ? 'مدير' : Math.random() > 0.4 ? 'مشرف' : 'مستخدم'
            };
            
            showNotification('تم تسجيل الدخول بنجاح!', 'success');
            hideLogin();
            showDashboard(user);
            
            // تحديث الإحصائيات
            let count = parseInt(document.getElementById('loginCount').textContent);
            document.getElementById('loginCount').textContent = count + 1;
            
            // بدء عداد الجلسة
            startSessionTimer();
        }

        // محاكاة التسجيل
        function simulateRegister() {
            const name = document.getElementById('regName').value;
            const email = document.getElementById('regEmail').value;
            const password = document.getElementById('regPassword').value;
            
            if (!name || !email || !password) {
                showNotification('يرجى ملء جميع الحقول', 'error');
                return;
            }
            
            showNotification('تم إنشاء الحساب بنجاح!', 'success');
            hideRegister();
            showLogin();
        }

        // قوة كلمة المرور
        function checkPasswordStrength(password) {
            let strength = 0;
            if (password.length >= 8) strength++;
            if (/[A-Z]/.test(password)) strength++;
            if (/[a-z]/.test(password)) strength++;
            if (/[0-9]/.test(password)) strength++;
            if (/[^A-Za-z0-9]/.test(password)) strength++;
            
            const bar = document.getElementById('strengthBar');
            const text = document.getElementById('strengthText');
            
            let color = '#f44336';
            let level = 'ضعيفة';
            
            if (strength >= 4) {
                color = '#4CAF50';
                level = 'قوية';
                bar.style.width = '100%';
            } else if (strength >= 3) {
                color = '#ff9800';
                level = 'متوسطة';
                bar.style.width = '66%';
            } else {
                bar.style.width = '33%';
            }
            
            bar.style.background = color;
            text.textContent = `قوة كلمة المرور: ${level}`;
        }

        // عداد الجلسة
        let sessionTimer;
        function startSessionTimer() {
            let seconds = 0;
            sessionTimer = setInterval(() => {
                seconds++;
                const minutes = Math.floor(seconds / 60);
                document.getElementById('sessionTime').textContent = minutes;
            }, 1000);
        }

        // تسجيل الخروج
        function logout() {
            clearInterval(sessionTimer);
            showNotification('تم تسجيل الخروج', 'info');
            document.getElementById('dashboard').style.display = 'none';
            document.querySelector('.hero').style.display = 'block';
            document.querySelector('.features').style.display = 'block';
        }

        // تغيير الخلفية
        function changeBackground() {
            const backgrounds = [
                'linear-gradient(-45deg, #667eea, #764ba2, #f093fb, #f5576c)',
                'linear-gradient(-45deg, #4facfe, #00f2fe, #43e97b, #38f9d7)',
                'linear-gradient(-45deg, #fa709a, #fee140, #ff9a9e, #fad0c4)',
                'linear-gradient(-45deg, #a18cd1, #fbc2eb, #fad0c4, #ffecd2)'
            ];
            
            const bg = document.querySelector('.background-animation');
            const randomBg = backgrounds[Math.floor(Math.random() * backgrounds.length)];
            bg.style.background = randomBg;
            
            showNotification('تم تغيير الخلفية', 'info');
        }

        // تبديل الثيم
        function toggleTheme() {
            document.body.classList.toggle('dark-theme');
            const icon = document.querySelector('.nav-btn i');
            icon.className = document.body.classList.contains('dark-theme') 
                ? 'fas fa-sun' 
                : 'fas fa-moon';
                
            showNotification('تم تغيير الثيم', 'info');
        }

        // إظهار/إخفاء كلمة المرور
        function togglePassword(inputId) {
            const input = document.getElementById(inputId);
            input.type = input.type === 'password' ? 'text' : 'password';
        }

        // الإشعارات
        function showNotification(message, type = 'info') {
            const notifications = document.getElementById('notifications');
            const notification = document.createElement('div');
            notification.className = `notification ${type}`;
            notification.innerHTML = `
                <i class="fas fa-${type === 'success' ? 'check' : type === 'error' ? 'times' : 'info'}-circle"></i>
                <span>${message}</span>
            `;
            notifications.appendChild(notification);
            
            setTimeout(() => {
                notification.remove();
            }, 3000);
        }

        // التمرير السلس
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function(e) {
                e.preventDefault();
                const target = document.querySelector(this.getAttribute('href'));
                if (target) {
                    target.scrollIntoView({ behavior: 'smooth' });
                }
            });
        });

        // التهيئة
        document.addEventListener('DOMContentLoaded', function() {
            createStars();
            hideLogin();
            hideRegister();
            document.getElementById('dashboard').style.display = 'none';
        });
